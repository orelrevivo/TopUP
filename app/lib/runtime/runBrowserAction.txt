  async #runBrowserAction(action: ActionState) {
    if (action.type !== 'browser') {
      unreachable('Expected browser action');
    }

    const { command, url, selector, text } = action as unknown as any; // typing from actions.ts

    try {
      // First ensure we have a session ID if not starting
      let sessionId = typeof window !== 'undefined' ? localStorage.getItem('browserbaseSessionId') : null;

      const res = await fetch('/api/browser', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          action: command,
          sessionId,
          url,
          selector,
          text
        })
      });

      if (!res.ok) {
        const error = await res.json();
        throw new Error(error.error || 'Failed to execute browser action');
      }

      const data = await res.json();

      if (command === 'start' && data.sessionId && data.debugUrl) {
        localStorage.setItem('browserbaseSessionId', data.sessionId);
        
        // Use a dynamic import to avoid circular dependencies if any
        const { workbenchStore } = await import('~/lib/stores/workbench');
        workbenchStore.browserDebugUrl.set(data.debugUrl);
        
        // Auto-switch to browser tab
        const { chatStore } = await import('~/lib/stores/chat');
        chatStore.setKey('showWorkbench', true);
        const { default: workbenchEventBus } = await import('~/components/workbench/Workbench.client');
        // Actually, workbench can be selected via an event or we can just hope the user clicks it.
      }
      
      // Store result if needed for LLM context, but for now we just succeed
    } catch (error: any) {
      logger.error('Browser action failed', error);
      throw error;
    }
  }
